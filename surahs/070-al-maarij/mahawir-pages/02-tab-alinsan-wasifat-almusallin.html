الإنسان الهلوع إلا المصلين آيات 19-35
نص الآيات
﴿إِنَّ ٱلۡإِنسَٰنَ خُلِقَ هَلُوعًا (١٩) إِذَا مَسَّهُ ٱلشَّرُّ جَزُوعٗا (٢٠) وَإِذَا مَسَّهُ ٱلۡخَيۡرُ مَنُوعًا (٢١) إِلَّا ٱلۡمُصَلِّينَ (٢٢) ٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ دَآئِمُونَ (٢٣) وَٱلَّذِينَ فِيٓ أَمۡوَٰلِهِمۡ حَقّٞ مَّعۡلُومٞ (٢٤) لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ (٢٥) وَٱلَّذِينَ يُصَدِّقُونَ بِيَوۡمِ ٱلدِّينِ (٢٦) وَٱلَّذِينَ هُم مِّنۡ عَذَابِ رَبِّهِم مُّشۡفِقُونَ (٢٧) إِنَّ عَذَابَ رَبِّهِمۡ غَيۡرُ مَأۡمُونٖ (٢٨) وَٱلَّذِينَ هُمۡ لِفُرُوجِهِمۡ حَٰفِظُونَ (٢٩) إِلَّا عَلَىٰٓ أَزۡوَٰجِهِمۡ أَوۡ مَا مَلَكَتۡ أَيۡمَٰنُهُمۡ فَإِنَّهُمۡ غَيۡرُ مَلُومِينَ (٣٠) فَمَنِ ٱبۡتَغَىٰ وَرَآءَ ذَٰلِكَ فَأُوْلَٰٓئِكَ هُمُ ٱلۡعَادُونَ (٣١) وَٱلَّذِينَ هُمۡ لِأَمَٰنَٰتِهِمۡ وَعَهۡدِهِمۡ رَٰعُونَ (٣٢) وَٱلَّذِينَ هُم بِشَهَٰدَٰتِهِمۡ قَآئِمُونَ (٣٣) وَٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ يُحَافِظُونَ (٣٤) أُوْلَٰٓئِكَ فِي جَنَّٰتٖ مُّكۡرَمُونَ (٣٥)﴾
معاني المفردات
هَلُوعًا— شديد الجزع عند الشدة، شديد المنع عند الرخاء، وهو طبع جبلي في الإنسان إلا من هذّبه الإيمان.
جَزُوعٗا— شديد الجزع والتسخط وقلة الصبر عند نزول المكروه.
مَنُوعًا— شديد المنع والبخل، يمسك الخير ولا يخرج منه شيئاً لغيره.
دَآئِمُونَ— مواظبون مستمرون على أداء الصلاة في أوقاتها دون تفريط.
حَقّٞ مَّعۡلُومٞ— نصيب محدد معلوم من المال، كالزكاة المفروضة وغيرها من حقوق الفقراء.
لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ— السائل: من يسأل الناس حاجته، والمحروم: الفقير المتعفف الذي لا يسأل فيُظن غنياً.
مُّشۡفِقُونَ— خائفون وجلون من عذاب الله رغم إيمانهم وطاعتهم، لا يأمنون مكر الله.
غَيۡرُ مَأۡمُونٖ— لا يأمن منه أحد، فحتى الصالحون يخافونه ولا يتكلون على أعمالهم.
لِفُرُوجِهِمۡ حَٰفِظُونَ— يصونون عوراتهم عن الحرام، ولا يقعون في الزنا أو الفواحش.
مَا مَلَكَتۡ أَيۡمَٰنُهُمۡ— ما يملكونه من الإماء بملك اليمين، وهو حكم كان جارياً في زمن نزول القرآن.
ٱلۡعَادُونَ— المتجاوزون لحدود الله المتعدون عليها بطلب ما هو حرام.
رَٰعُونَ— حافظون مراعون لأماناتهم وعهودهم، لا يخونون ولا يخلفون.
المعنى الإجمالي
ينتقل المحور الثاني من السورة لبيان طبيعة الإنسان الجبلية: أنه خُلق هلوعاً، شديد الجزع عند نزول الشر، شديد المنع والبخل عند حصول الخير، وهذا طبع فطري في الإنسان قبل أن يهذّبه الإيمان. ثم تستثني الآيات المصلين من هذا الطبع الهلوع، وتُفصّل صفاتهم الجليلة تفصيلاً دقيقاً يبيّن حقيقة الإيمان العملي. فمن صفاتهم: المداومة على الصلاة دون تفريط، وإخراج حق معلوم من أموالهم للسائل والمحروم المتعفف، والتصديق الجازم بيوم الحساب، والخوف الدائم من عذاب الله رغم طاعتهم، وحفظ الفروج عن الحرام إلا في الحلال المشروع، ومن تجاوز ذلك فهو من المعتدين. ومن صفاتهم أيضاً: رعاية الأمانات والعهود، والقيام بالشهادة بالحق دون تحريف، والمحافظة الدائمة على الصلاة. وختمت الآيات بأن هؤلاء الموصوفين بهذه الصفات هم أهل الجنات المكرمون عند الله.
التدبر
المعنى التفصيلي
إِنَّ ٱلۡإِنسَٰنَ خُلِقَ هَلُوعًا (١٩) إِذَا مَسَّهُ ٱلشَّرُّ جَزُوعٗا (٢٠) وَإِذَا مَسَّهُ ٱلۡخَيۡرُ مَنُوعًا (٢١)
وصف عام لطبيعة الإنسان الجبلية قبل أن يهذّبها الإيمان: أنه هلوع، شديد الجزع والتسخط عند نزول المكروه، شديد المنع والبخل عند حصول النعمة والغنى، وهو تشخيص قرآني دقيق لضعف النفس البشرية.
إِلَّا ٱلۡمُصَلِّينَ (٢٢) ٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ دَآئِمُونَ (٢٣)
استثناء المصلين من هذا الطبع الهلوع، وأول صفاتهم: المداومة على الصلاة والمواظبة عليها في أوقاتها بخشوع وانتظام، فالصلاة هي الأساس الذي يُهذّب النفس ويقيها من الهلع.
وَٱلَّذِينَ فِيٓ أَمۡوَٰلِهِمۡ حَقّٞ مَّعۡلُومٞ (٢٤) لِّلسَّآئِلِ وَٱلۡمَحۡرُومِ (٢٥)
الصفة الثانية: إخراج حق معلوم من أموالهم يُصرف للسائل الذي يسأل الناس حاجته، وللمحروم الذي يتعفف عن السؤال فيُظن غنياً وهو فقير، وهذا نقيض المنع المذكور في طبع الإنسان الهلوع.
وَٱلَّذِينَ يُصَدِّقُونَ بِيَوۡمِ ٱلدِّينِ (٢٦)
الصفة الثالثة: التصديق الجازم بيوم القيامة والحساب، وهذا التصديق هو الدافع الحقيقي وراء الالتزام بباقي الصفات المذكورة، فمن آمن بالحساب استعد له.
وَٱلَّذِينَ هُم مِّنۡ عَذَابِ رَبِّهِم مُّشۡفِقُونَ (٢٧) إِنَّ عَذَابَ رَبِّهِمۡ غَيۡرُ مَأۡمُونٖ (٢٨)
الصفة الرابعة: الخوف الدائم من عذاب الله رغم طاعتهم وعبادتهم، فهم لا يتكلون على أعمالهم ولا يأمنون مكر الله، لأن عذاب الله لا يأمن منه أحد مهما بلغ من الصلاح.
وَٱلَّذِينَ هُمۡ لِفُرُوجِهِمۡ حَٰفِظُونَ (٢٩) إِلَّا عَلَىٰٓ أَزۡوَٰجِهِمۡ أَوۡ مَا مَلَكَتۡ أَيۡمَٰنُهُمۡ فَإِنَّهُمۡ غَيۡرُ مَلُومِينَ (٣٠) فَمَنِ ٱبۡتَغَىٰ وَرَآءَ ذَٰلِكَ فَأُوْلَٰٓئِكَ هُمُ ٱلۡعَادُونَ (٣١)
الصفة الخامسة: حفظ الفروج عن الحرام كلياً، إلا في الحلال المشروع من الزواج وملك اليمين، فمن تجاوز هذين الحدين طالباً غيرهما فهو من المعتدين المتجاوزين لحدود الله.
وَٱلَّذِينَ هُمۡ لِأَمَٰنَٰتِهِمۡ وَعَهۡدِهِمۡ رَٰعُونَ (٣٢) وَٱلَّذِينَ هُم بِشَهَٰدَٰتِهِمۡ قَآئِمُونَ (٣٣)
الصفة السادسة والسابعة: رعاية الأمانات وعدم خيانتها، والوفاء بالعهود وعدم نقضها، والقيام بالشهادة بالحق كما هي دون تحريف أو كتمان، وهي من أعظم صفات الصدق والاستقامة.
وَٱلَّذِينَ هُمۡ عَلَىٰ صَلَاتِهِمۡ يُحَافِظُونَ (٣٤) أُوْلَٰٓئِكَ فِي جَنَّٰتٖ مُّكۡرَمُونَ (٣٥)
ختام الصفات بتكرار المحافظة على الصلاة (بعد ذكر المداومة عليها في البداية)، وهذا التكرار يؤكد أهمية الصلاة كإطار جامع لكل الصفات المذكورة، ثم يأتي الجزاء: أن هؤلاء الموصوفين بكل هذه الصفات هم أهل الجنات المكرمون عند الله.
